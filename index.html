⚙️ Google Apps Script (GAS) 代碼範例：

請複製以下代碼貼入 Google 試算表 Apps Script 編輯器中並部署為 Web App：

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // 若工作表為空則建立表頭
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "時間戳記", "班別", "學生姓名", "題目ID", "題目內容", 
        "學生答案", "正確答案", "是否正確", "知識層面/薄弱原因"
      ]);
    }
    
    // 遍歷紀錄寫入
    if (data.records && data.records.length > 0) {
      data.records.forEach(function(r) {
        sheet.appendRow([
          new Date(), data.studentClass, data.studentName,
          r.questionId, r.questionText, r.userAnswer,
          r.correctAnswer, r.isCorrect ? "正確" : "錯誤", r.knowledgeTag
        ]);
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
